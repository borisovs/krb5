cmake_minimum_required(VERSION 3.20)

# Generate *.h/*.c  MIT Kerberos5 files
find_program(AWK_BIN NAMES awk gawk)
if(NOT AWK_BIN)
    message(FATAL_ERROR "awk not found. Please install awk or set the AWK_BIN variable manually.")
endif()

find_program(CAT_BIN NAMES cat)
if(NOT CAT_BIN)
    message(FATAL_ERROR "cat not found. Please install cat or set the CAT_BIN variable manually.")
endif()

set(AH  ${ROOT_DIR}/util/et/et_h.awk)
set(AC  ${ROOT_DIR}/util/et/et_c.awk)
set(INC ${ROOT_DIR}/include)
set(ET  ${ROOT_DIR}/lib/krb5/error_tables)
set(GG  ${ROOT_DIR}/lib/gssapi/generic)
set(GK  ${ROOT_DIR}/lib/gssapi/krb5)
set(PR  ${ROOT_DIR}/util/profile)
set(CE  ${ROOT_DIR}/util/et)
set(CCL ${ROOT_DIR}/ccapi/lib)

set(KRBHDEP ${INC}/krb5/krb5.hin
            ${INC}/krb5_err.h
            ${INC}/k5e1_err.h
            ${INC}/kdb5_err.h
            ${INC}/kv5m_err.h
            ${INC}/krb524_err.h
            ${INC}/asn1_err.h
)

set(ETOUT ${INC}/asn1_err.h
          ${ET}/asn1_err.c
          ${INC}/kdb5_err.h
          ${ET}/kdb5_err.c
          ${INC}/krb5_err.h
          ${ET}/krb5_err.c
          ${INC}/k5e1_err.h
          ${ET}/k5e1_err.c
          ${INC}/kv5m_err.h
          ${ET}/kv5m_err.c
          ${INC}/krb524_err.h
          ${ET}/krb524_err.c
          ${PR}/prof_err.h
          ${PR}/prof_err.c
          ${GG}/gssapi_err_generic.h
          ${GG}/gssapi_err_generic.c
          ${GK}/gssapi_err_krb5.h
          ${GK}/gssapi_err_krb5.c
          ${CCL}/ccapi_err.h
          ${CCL}/ccapi_err.c
)

set(HOUT ${INC}/krb5/krb5.h
         ${GG}/gssapi.h
         ${PR}/profile.h
)

add_custom_command(OUTPUT ${INC}/asn1_err.h
    COMMAND ${AWK_BIN} -f ${AH} outfile=${INC}/asn1_err.h ${ET}/asn1_err.et
    COMMENT "Generating asn1_err.h"
)

    COMMAND ${CMAKE_COMMAND} -E rm -f ${INC}/krb5/krb5.h
    COMMAND ${CAT_BIN} ${KRBHDEP} > ${INC}/krb5/krb5.h
    COMMENT "Creating ${INC}/krb5/krb5.h"
)


add_custom_command(OUTPUT ${PR}/profile.h
    COMMAND ${CMAKE_COMMAND} -E rm -f ${PR}/profile.h
    COMMAND ${CAT_BIN} ${PR}/profile.hin  ${PR}/prof_err.h > ${PR}/profile.h
    COMMENT "Creating ${PR}/profile.h"
)

add_custom_command(OUTPUT ${GG}/gssapi.h
    COMMAND ${CMAKE_COMMAND} -E rm -f ${GG}/gssapi.h
    COMMAND ${CAT_BIN} ${GG}/gssapi.hin > ${PR}/gssapi.h
    COMMENT "Creating ${GG}/gssapi.h"
)
